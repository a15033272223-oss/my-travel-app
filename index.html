        <div id="equipment" class="tab-content active bg-white p-4 rounded-xl shadow-sm border border-gray-100">

            <h2 class="text-lg font-bold text-slate-700 mb-4">行前裝備檢查</h2>

    <script>

        // --- 頁籤切換邏輯 ---

        function switchTab(tabId) {

            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));

            document.querySelectorAll('.tab-btn').forEach(el => {

                el.classList.remove('text-blue-600', 'border-b-2', 'border-blue-600');

                el.classList.add('text-gray-500');

            });

            document.getElementById(tabId).classList.add('active');

            const activeBtn = document.getElementById('btn-' + tabId);

            activeBtn.classList.remove('text-gray-500');

            activeBtn.classList.add('text-blue-600', 'border-b-2', 'border-blue-600');

        }


        // --- 動態裝備清單邏輯 ---

        

        // 1. 從瀏覽器的 localStorage 讀取資料，如果沒有，就載入預設值

        let equipmentData = JSON.parse(localStorage.getItem('myTripEquipment')) || [

            { id: 1, text: 'DJI Action 5 Pro 與防水殼', checked: false },

            { id: 2, text: '面鏡與長蛙鞋', checked: false },

            { id: 3, text: '個人藥品與護照', checked: false }

        ];


        // 2. 將資料渲染(畫)到網頁上

        function renderEquipment() {

            const list = document.getElementById('equipmentList');

            list.innerHTML = ''; // 先清空畫面

            

            equipmentData.forEach(item => {

                list.innerHTML += `

                    <label class="flex items-center space-x-3 cursor-pointer p-2 hover:bg-slate-50 rounded-lg transition-colors border border-transparent hover:border-slate-100">

                        <input type="checkbox" class="w-5 h-5 text-blue-600 rounded" 

                               ${item.checked ? 'checked' : ''} 

                               onchange="toggleItem(${item.id})">

                        <span class="text-slate-700 ${item.checked ? 'checked-item' : ''}">${item.text}</span>

                        <button onclick="deleteItem(${item.id})" class="ml-auto text-xs text-red-400 hover:text-red-600 hidden md:block">刪除</button>

                    </label>

                `;

            });

        }


        // 3. 新增項目的功能

        function addItem() {

            const input = document.getElementById('newItemInput');

            const text = input.value.trim(); // 取得輸入內容並去除前後空白

            

            if (text !== '') {

                // 給新項目一個唯一的 ID (用當下時間戳記)

                equipmentData.push({ id: Date.now(), text: text, checked: false });

                input.value = ''; // 清空輸入框

                saveData();       // 存檔

                renderEquipment(); // 重新整理畫面

            }

        }


        // 4. 打勾/取消打勾的功能

        function toggleItem(id) {

            // 找出被點擊的那個項目

            const item = equipmentData.find(i => i.id === id);

            if (item) {

                item.checked = !item.checked; // 狀態反轉

                saveData();

                renderEquipment();

            }

        }


        // 5. 刪除項目的功能

        function deleteItem(id) {

            // 過濾掉要刪除的 ID，保留其他項目

            equipmentData = equipmentData.filter(i => i.id !== id);

            saveData();

            renderEquipment();

        }


        // 6. 存檔到 localStorage

        function saveData() {

            localStorage.setItem('myTripEquipment', JSON.stringify(equipmentData));

        }


        // 網頁載入時，執行第一次畫面渲染

        renderEquipment();

        

        // 讓輸入框按下 Enter 也能新增

        document.getElementById('newItemInput').addEventListener('keypress', function (e) {

            if (e.key === 'Enter') addItem();

        });

    </script>
